type ChartType = "line" | "candle";
type IndicatorType = "MA" | "EMA" | "BB" | "VOL" | "RSI" | "MACD";

interface PriceChartProps {
  data: any[];
  symbol: string;
}

export function PriceChart({ data, symbol }: PriceChartProps) {
  const chartContainerRef = useRef<HTMLDivElement>(null);
  const chartRef = useRef<any>(null);
  const mainSeriesRef = useRef<any>(null);
  const indicatorSeriesRefs = useRef<Record<string, any[]>>({});
  const rsiChartRef = useRef<any>(null);
  const macdChartRef = useRef<any>(null);
  const rsiContainerRef = useRef<HTMLDivElement>(null);
  const macdContainerRef = useRef<HTMLDivElement>(null);

  const [chartType, setChartType] = useState<ChartType>("line");
  const [activeIndicators, setActiveIndicators] = useState<Set<IndicatorType>>(new Set());
  const [crosshairPrice, setCrosshairPrice] = useState<string | null>(null);

  const timeframes = ["1D", "1W", "1M", "3M", "1Y", "YTD"];
  const [activeTimeframe, setActiveTimeframe] = useState("1D");

  // Map raw data to OHLCV format
  // If data only has close prices, simulate OHLCV
  const candleData = (data || []).map((d: any, i: number) => {
    const close = d.close ?? d.value ?? d.price ?? 0;
    const open = d.open ?? (i > 0 ? (data[i - 1].close ?? data[i - 1].value ?? close) : close * 0.998);
    const high = d.high ?? Math.max(open, close) * (1 + Math.random() * 0.005);
    const low = d.low ?? Math.min(open, close) * (1 - Math.random() * 0.005);
    const volume = d.volume ?? Math.floor(Math.random() * 10000000 + 1000000);
    return {
      time: d.time ?? d.date ?? d.timestamp,
      open: parseFloat(open.toFixed(0)),
      high: parseFloat(high.toFixed(0)),
      low: parseFloat(low.toFixed(0)),
      close: parseFloat(close.toFixed(0)),
      value: parseFloat(close.toFixed(0)),
      volume,
    };
  }).filter(d => d.time);

  const calcMA = (period: number) => {
    return candleData.map((d, i) => {
      if (i < period - 1) return null;
      const avg = candleData.slice(i - period + 1, i + 1).reduce((s, x) => s + x.close, 0) / period;
      return { time: d.time, value: parseFloat(avg.toFixed(2)) };
    }).filter(Boolean);
  };

  const calcEMA = (period: number) => {
    const k = 2 / (period + 1);
    let ema = candleData[0]?.close ?? 0;
    return candleData.map((d, i) => {
      if (i === 0) { ema = d.close; return { time: d.time, value: ema }; }
      ema = d.close * k + ema * (1 - k);
      return { time: d.time, value: parseFloat(ema.toFixed(2)) };
    });
  };

  const calcBB = (period = 20, stdDev = 2) => {
    const mid = calcMA(period);
    return mid.map((m: any, i: number) => {
      const slice = candleData.slice(i, i + period);
      const avg = m.value;
      const variance = slice.reduce((s, x) => s + Math.pow(x.close - avg, 2), 0) / period;
      const sd = Math.sqrt(variance) * stdDev;
      return {
        time: m.time,
        upper: parseFloat((avg + sd).toFixed(2)),
        mid: avg,
        lower: parseFloat((avg - sd).toFixed(2)),
      };
    });
  };

  const calcRSI = (period = 14) => {
    let gains = 0, losses = 0;
    return candleData.map((d, i) => {
      if (i === 0) return { time: d.time, value: 50 };
      const diff = d.close - candleData[i - 1].close;
      if (i <= period) { gains += diff > 0 ? diff : 0; losses += diff < 0 ? -diff : 0; }
      else {
        gains = (gains * (period - 1) + (diff > 0 ? diff : 0)) / period;
        losses = (losses * (period - 1) + (diff < 0 ? -diff : 0)) / period;
      }
      const rs = losses === 0 ? 100 : gains / losses;
      return { time: d.time, value: parseFloat((100 - 100 / (1 + rs)).toFixed(2)) };
    });
  };

  useEffect(() => {
    if (!chartContainerRef.current || candleData.length === 0) return;

    const chart = createChart(chartContainerRef.current, {
      layout: {
        background: { color: "#0f0f0f" },
        textColor: "#888",
        fontFamily: "Inter, sans-serif",
        fontSize: 11,
      },
      grid: {
        vertLines: { color: "#1a1a1a" },
        horzLines: { color: "#1a1a1a" },
      },
      crosshair: {
        mode: CrosshairMode.Normal,
        vertLine: { color: "#444", width: 1, style: LineStyle.Dashed, labelBackgroundColor: "#222" },
        horzLine: { color: "#444", width: 1, style: LineStyle.Dashed, labelBackgroundColor: "#222" },
      },
      rightPriceScale: { borderColor: "#222", scaleMargins: { top: 0.1, bottom: 0.2 } },
      timeScale: { borderColor: "#222", timeVisible: true, secondsVisible: false },
      width: chartContainerRef.current.clientWidth,
      height: 340,
    });
    chartRef.current = chart;

    const resizeObs = new ResizeObserver(entries => {
      if (entries[0]) chart.applyOptions({ width: entries[0].contentRect.width });
    });
    resizeObs.observe(chartContainerRef.current);

    // Main series
    let mainSeries: any;
    if (chartType === "candle") {
      mainSeries = chart.addSeries(CandlestickSeries, {
        upColor: "#26a69a", downColor: "#ef5350",
        borderUpColor: "#26a69a", borderDownColor: "#ef5350",
        wickUpColor: "#26a69a", wickDownColor: "#ef5350",
      });
      mainSeries.setData(candleData.map(d => ({ time: d.time, open: d.open, high: d.high, low: d.low, close: d.close })));
    } else {
      mainSeries = chart.addSeries(LineSeries, { color: "#38BDF8", lineWidth: 2, priceLineVisible: false });
      mainSeries.setData(candleData.map(d => ({ time: d.time, value: d.value })));
    }
    mainSeriesRef.current = mainSeries;

    // Volume
    const volSeries = chart.addSeries(HistogramSeries, {
      color: "#1e3a5f", priceFormat: { type: "volume" },
      priceScaleId: "vol",
    });
    chart.priceScale("vol").applyOptions({ scaleMargins: { top: 0.8, bottom: 0 } });
    volSeries.setData(candleData.map(d => ({
      time: d.time,
      value: d.volume,
      color: d.close >= d.open ? "#26a69a44" : "#ef535044",
    })));

    // Indicators
    const indicatorSeries: Record<string, any[]> = {};
    if (activeIndicators.has("MA")) {
      const ma20 = chart.addSeries(LineSeries, { color: "#f59e0b", lineWidth: 1, priceLineVisible: false });
      ma20.setData(calcMA(20) as any);
      const ma50 = chart.addSeries(LineSeries, { color: "#8b5cf6", lineWidth: 1, priceLineVisible: false });
      ma50.setData(calcMA(50) as any);
      indicatorSeries["MA"] = [ma20, ma50];
    }
    if (activeIndicators.has("EMA")) {
      const ema9 = chart.addSeries(LineSeries, { color: "#34d399", lineWidth: 1, priceLineVisible: false, lineStyle: LineStyle.Dashed });
      ema9.setData(calcEMA(9) as any);
      const ema21 = chart.addSeries(LineSeries, { color: "#f97316", lineWidth: 1, priceLineVisible: false, lineStyle: LineStyle.Dashed });
      ema21.setData(calcEMA(21) as any);
      indicatorSeries["EMA"] = [ema9, ema21];
    }
    if (activeIndicators.has("BB")) {
      const bb = calcBB();
      const bbUpper = chart.addSeries(LineSeries, { color: "#6366f1", lineWidth: 1, priceLineVisible: false, lineStyle: LineStyle.Dotted });
      bbUpper.setData(bb.map((b: any) => ({ time: b.time, value: b.upper })));
      const bbMid = chart.addSeries(LineSeries, { color: "#6366f1", lineWidth: 1, priceLineVisible: false });
      bbMid.setData(bb.map((b: any) => ({ time: b.time, value: b.mid })));
      const bbLower = chart.addSeries(LineSeries, { color: "#6366f1", lineWidth: 1, priceLineVisible: false, lineStyle: LineStyle.Dotted });
      bbLower.setData(bb.map((b: any) => ({ time: b.time, value: b.lower })));
      indicatorSeries["BB"] = [bbUpper, bbMid, bbLower];
    }
    indicatorSeriesRefs.current = indicatorSeries;

    chart.subscribeCrosshairMove(param => {
      if (param.point && mainSeries) {
        const price = param.seriesData.get(mainSeries);
        if (price) {
          const val = (price as any).close ?? (price as any).value;
          setCrosshairPrice(val ? parseInt(val).toLocaleString("id-ID") : null);
        }
      }
    });

    chart.timeScale().fitContent();

    return () => { resizeObs.disconnect(); chart.remove(); };
  }, [chartType, activeIndicators, data]);

  // RSI sub-chart
  useEffect(() => {
    if (!rsiContainerRef.current || !activeIndicators.has("RSI") || candleData.length === 0) return;
    const chart = createChart(rsiContainerRef.current, {
      layout: { background: { color: "#0f0f0f" }, textColor: "#888", fontSize: 10 },
      grid: { vertLines: { color: "#1a1a1a" }, horzLines: { color: "#1a1a1a" } },
      rightPriceScale: { borderColor: "#222", scaleMargins: { top: 0.1, bottom: 0.1 } },
      timeScale: { borderColor: "#222", timeVisible: true, visible: false },
      crosshair: { mode: CrosshairMode.Normal },
      width: rsiContainerRef.current.clientWidth,
      height: 100,
    });
    rsiChartRef.current = chart;
    const rsiSeries = chart.addSeries(LineSeries, { color: "#a78bfa", lineWidth: 1, priceLineVisible: false });
    rsiSeries.setData(calcRSI() as any);
    const obs = new ResizeObserver(e => { if (e[0]) chart.applyOptions({ width: e[0].contentRect.width }); });
    obs.observe(rsiContainerRef.current);
    chart.timeScale().fitContent();
    return () => { obs.disconnect(); chart.remove(); };
  }, [activeIndicators, data]);

  const toggleIndicator = (ind: IndicatorType) => {
    setActiveIndicators(prev => {
      const next = new Set(prev);
      if (next.has(ind)) next.delete(ind); else next.add(ind);
      return next;
    });
  };

  const indicators: { key: IndicatorType; label: string; color: string }[] = [
    { key: "MA", label: "MA", color: "#f59e0b" },
    { key: "EMA", label: "EMA", color: "#34d399" },
    { key: "BB", label: "BB", color: "#6366f1" },
    { key: "RSI", label: "RSI", color: "#a78bfa" },
  ];

  return (
    <div className="rounded-lg border border-border/40 bg-[#0f0f0f] overflow-hidden">
      {/* Toolbar */}
      <div className="flex items-center justify-between gap-2 px-4 py-2 border-b border-border/30">
        <div className="flex items-center gap-1">
          <span className="text-xs text-muted-foreground font-semibold mr-2 uppercase tracking-wider">Grafik</span>
          {/* Chart type toggle */}
          <button
            onClick={() => setChartType("line")}
            className={`px-2.5 py-1 rounded text-xs font-semibold transition-colors ${chartType === "line" ? "bg-[#38BDF8] text-black" : "text-muted-foreground hover:text-foreground bg-transparent"}`}
          >
            Line
          </button>
          <button
            onClick={() => setChartType("candle")}
            className={`px-2.5 py-1 rounded text-xs font-semibold transition-colors ${chartType === "candle" ? "bg-[#38BDF8] text-black" : "text-muted-foreground hover:text-foreground bg-transparent"}`}
          >
            Candle
          </button>
        </div>

        {/* Timeframes */}
        <div className="flex items-center gap-0.5">
          {timeframes.map(tf => (
            <button
              key={tf}
              onClick={() => setActiveTimeframe(tf)}
              className={`px-2 py-1 rounded text-xs font-semibold transition-colors ${activeTimeframe === tf ? "bg-primary text-primary-foreground" : "text-muted-foreground hover:text-foreground"}`}
            >
              {tf}
            </button>
          ))}
        </div>
      </div>

      {/* Indicator toolbar */}
      <div className="flex items-center gap-2 px-4 py-1.5 border-b border-border/20 bg-black/30">
        <span className="text-[10px] text-muted-foreground uppercase tracking-widest mr-1">Indikator</span>
        {indicators.map(ind => (
          <button
            key={ind.key}
            onClick={() => toggleIndicator(ind.key)}
            className={`px-2 py-0.5 rounded text-[10px] font-bold border transition-all ${
              activeIndicators.has(ind.key)
                ? "text-black border-transparent"
                : "border-border/40 text-muted-foreground hover:text-foreground hover:border-foreground/30 bg-transparent"
            }`}
            style={activeIndicators.has(ind.key) ? { backgroundColor: ind.color, borderColor: ind.color } : {}}
          >
            {ind.label}
          </button>
        ))}
        {crosshairPrice && (
          <span className="ml-auto text-xs font-mono text-[#38BDF8]">IDR {crosshairPrice}</span>
        )}
      </div>

      {/* Main chart */}
      <div ref={chartContainerRef} className="w-full" style={{ height: 340 }} />

      {/* RSI sub-chart */}
      {activeIndicators.has("RSI") && (
        <div className="border-t border-border/20">
          <div className="px-4 py-1 flex items-center gap-2">
            <span className="text-[9px] text-muted-foreground font-bold uppercase tracking-widest">RSI (14)</span>
            <span className="text-[9px] text-muted-foreground">— Overbought: 70 | Oversold: 30</span>
          </div>
          <div ref={rsiContainerRef} className="w-full" style={{ height: 100 }} />
        </div>
      )}
    </div>
  );
}
