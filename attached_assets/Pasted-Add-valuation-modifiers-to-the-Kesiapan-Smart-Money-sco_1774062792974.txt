PART 1 — Create server/engine/valuationModifier.ts

/**
 * ============================================================
 * VALUATION MODIFIER v1.0
 * ============================================================
 * Applies a small valuation + quality adjustment to the
 * v1.1 bandarmology readiness score.
 *
 * DESIGN RULES:
 *   - Bandarmology remains dominant (90%+ of signal)
 *   - Max total adjustment: ±5 points
 *   - Deterministic: same input → same output
 *   - Safe: null inputs → 0 modifier (no adjustment)
 *   - Never pushes a HINDARI stock into WATCHLIST range
 *   - Never pushes a WATCHLIST stock into SIAP range
 *
 * MODIFIER TABLE:
 *   Valuation:  MURAH=+3, WAJAR=0, MAHAL=-3, NO_DATA=0
 *   Quality:    KUAT=+2,  SEDANG=0, LEMAH=-2, NO_DATA=0
 *   Max total:  clamp(sum, -5, +5)
 * ============================================================
 */

export type ValuationLabel = 'MURAH' | 'WAJAR' | 'MAHAL' | 'TIDAK_ADA_DATA';
export type QualityLabel   = 'KUAT'  | 'SEDANG' | 'LEMAH' | 'TIDAK_ADA_DATA';

export interface ValuationModifierInput {
  valuationLabel: ValuationLabel | null | undefined;
  qualityLabel:   QualityLabel   | null | undefined;
  currentReadiness: number;
}

export interface ValuationModifierOutput {
  adjustedReadiness: number;
  modifier: number;           // the actual ±N applied
  valuationComponent: number; // -3, 0, or +3
  qualityComponent: number;   // -2, 0, or +2
  wasAdjusted: boolean;
}

export function applyValuationModifier(
  input: ValuationModifierInput
): ValuationModifierOutput {
  const { valuationLabel, qualityLabel, currentReadiness } = input;

  // Safety: invalid readiness → return unchanged
  if (
    currentReadiness === null ||
    currentReadiness === undefined ||
    isNaN(currentReadiness) ||
    !isFinite(currentReadiness)
  ) {
    return {
      adjustedReadiness: currentReadiness ?? 0,
      modifier: 0,
      valuationComponent: 0,
      qualityComponent: 0,
      wasAdjusted: false,
    };
  }

  // Valuation component
  let valuationComponent = 0;
  if (valuationLabel === 'MURAH') valuationComponent = +3;
  if (valuationLabel === 'MAHAL') valuationComponent = -3;

  // Quality component
  let qualityComponent = 0;
  if (qualityLabel === 'KUAT')  qualityComponent = +2;
  if (qualityLabel === 'LEMAH') qualityComponent = -2;

  // Total modifier: clamp to ±5
  const rawModifier = valuationComponent + qualityComponent;
  const modifier = Math.max(-5, Math.min(5, rawModifier));

  // Apply modifier
  let adjusted = currentReadiness + modifier;

  // BOUNDARY PROTECTION:
  // Do not cross bucket thresholds purely due to valuation modifier.
  // This prevents a borderline HINDARI (59) becoming WATCHLIST (60)
  // just because valuation is MURAH.
  // Thresholds: <60=HINDARI, 60-79=WATCHLIST, ≥80=SIAP
  const originalBucket =
    currentReadiness >= 80 ? 'SIAP' :
    currentReadiness >= 60 ? 'WATCHLIST' : 'HINDARI';

  const adjustedBucket =
    adjusted >= 80 ? 'SIAP' :
    adjusted >= 60 ? 'WATCHLIST' : 'HINDARI';

  // If modifier would push across a bucket boundary, clamp at boundary
  if (originalBucket !== adjustedBucket) {
    if (originalBucket === 'HINDARI' && adjustedBucket === 'WATCHLIST') {
      adjusted = 59; // cap just below WATCHLIST threshold
    } else if (originalBucket === 'WATCHLIST' && adjustedBucket === 'SIAP') {
      adjusted = 79; // cap just below SIAP threshold
    } else if (originalBucket === 'WATCHLIST' && adjustedBucket === 'HINDARI') {
      adjusted = 60; // floor just above HINDARI threshold
    } else if (originalBucket === 'SIAP' && adjustedBucket === 'WATCHLIST') {
      adjusted = 80; // floor just above WATCHLIST threshold
    }
  }

  // Final clamp 0-100
  adjusted = Math.max(0, Math.min(100, Math.round(adjusted)));

  return {
    adjustedReadiness: adjusted,
    modifier,
    valuationComponent,
    qualityComponent,
    wasAdjusted: modifier !== 0,
  };
}

PART 2 — Wire modifier into routes.ts

Find the GET /api/stock/:symbol endpoint in routes.ts.
This is the endpoint that returns the full stock object
with readiness score.

After the existing readiness score is computed (from 
getStockDecision() in unifiedDecision.ts), add this block:

// Apply valuation modifier to readiness score
try {
  const { computeValuation } = await import('./engine/valuationEngine');
  const { applyValuationModifier } = await import('./engine/valuationModifier');

  const valuationOutput = computeValuation({
    peRatio:       stockData.peRatio       ?? null,
    dividendYield: stockData.dividendYield ?? null,
    roe:           stockData.roe           ?? null,
    netMargin:     stockData.netMargin     ?? null,
    sector:        stockData.sector        ?? null,
  });

  const modifierResult = applyValuationModifier({
    valuationLabel:   valuationOutput.valuation.label,
    qualityLabel:     valuationOutput.quality.label,
    currentReadiness: stockData.readiness ?? 0,
  });

  if (modifierResult.wasAdjusted) {
    stockData.readiness = modifierResult.adjustedReadiness;
    stockData.valuationModifier = modifierResult.modifier;
    stockData.valuationLabel    = valuationOutput.valuation.label;
    stockData.qualityLabel      = valuationOutput.quality.label;
  }
} catch (err) {
  // Valuation modifier is non-critical — never break the main response
  console.error('[/api/stock] Valuation modifier error:', err);
}

IMPORTANT: Find the exact location in routes.ts where the 
stock response object is assembled. The modifier must run 
AFTER getStockDecision() and BEFORE the response is sent.
If stockData is immutable, create a mutable copy first.

Also apply the same modifier to GET /api/stocks (the bulk
endpoint) so the homepage and radar scores are consistent.
Use the same try/catch pattern — non-critical, never throws.
